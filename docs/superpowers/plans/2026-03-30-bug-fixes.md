```python
# utils.py
"""Shared utilities for X-tracker modules."""
import asyncio, json, os, sqlite3, sys
from pathlib import Path
from typing import Optional

import httpx
import yaml
from dotenv import load_dotenv

_DEFAULT_BASE = Path(__file__).parent
load_dotenv(_DEFAULT_BASE / ".env")


def load_account_config(account_name: str, base_path: Path = _DEFAULT_BASE) -> dict:
    """Load account config from accounts.yaml. sys.exit(1) if account not found."""
    config_path = base_path / "accounts.yaml"
    with open(config_path) as f:
        data = yaml.safe_load(f)
    accounts = data.get("accounts", {})
    if account_name not in accounts:
        print(f"Error: account '{account_name}' not found in accounts.yaml")
        print(f"Available: {list(accounts.keys())}")
        sys.exit(1)
    cfg = accounts[account_name]
    cfg["username"] = account_name
    cfg["discord_webhook"] = os.environ.get(cfg.get("discord_webhook_env", ""), "")
    return cfg


def get_db_conn(db_path: "str | Path") -> sqlite3.Connection:
    """Open SQLite connection with WAL mode and row_factory=Row."""
    conn = sqlite3.connect(str(db_path))
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    return conn


async def send_discord(
    webhook: str,
    content: str,
    image_paths: Optional[list] = None,
) -> None:
    """Send message to Discord webhook.

    - image_paths: attach up to 4 images.
    - Long text (>1900 chars) is automatically split into chunks.
    - Callers are responsible for including tweet_url inside content if needed.
    """
    if not webhook:
        print("  [skip] Discord webhook not configured")
        return

    async with httpx.AsyncClient() as client:
        if image_paths:
            opened = []
            try:
                files = {}
                for i, p in enumerate(image_paths[:4]):
                    f = open(p, "rb")
                    opened.append(f)
                    files[f"file{i}"] = (Path(p).name, f, "image/jpeg")
                files["payload_json"] = (None, json.dumps({"content": content}), "application/json")
                await client.post(webhook, files=files)
            finally:
                for f in opened:
                    f.close()
        else:
            chunks: list[str] = []
            remaining = content
            while len(remaining) > 1900:
                idx = remaining.rfind("\n", 0, 1900)
                if idx == -1:
                    idx = 1900
                chunks.append(remaining[:idx])
                remaining = remaining[idx:].strip()
            chunks.append(remaining)
            for chunk in chunks:
                if chunk:
                    await client.post(webhook, json={"content": chunk})
                    await asyncio.sleep(1)
```

def init_db(db_path=None):
    """Create tables if not exist. db_path=None uses module-level DB_PATH."""
    path = db_path or DB_PATH
    conn = sqlite3.connect(str(path))
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("""
        CREATE TABLE IF NOT EXISTS tweets (
            id TEXT PRIMARY KEY,
            account TEXT NOT NULL,
            created_at TEXT,
            text TEXT,
            images TEXT,
            scraped_at TEXT
        )
    """)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS query_cache (
            topic TEXT PRIMARY KEY,
            result_json TEXT NOT NULL,
            updated_at TEXT NOT NULL
        )
    """)
    try:
        conn.execute(
            "CREATE VIRTUAL TABLE IF NOT EXISTS tweets_fts "
            "USING fts5(text, content='tweets', content_rowid='rowid')"
        )
    except sqlite3.OperationalError:
        pass
    conn.commit()
    return conn
```

def search_tweets_fts(conn, account: str, topic: str, days: int) -> list:
    """Search tweets using FTS5 index. Falls back to LIKE if FTS unavailable."""
    from datetime import datetime, timedelta
    since_date = (datetime.now() - timedelta(days=days)).isoformat()
    try:
        rows = conn.execute(
            """SELECT tweets.id, tweets.created_at, tweets.text
               FROM tweets_fts
               JOIN tweets ON tweets.rowid = tweets_fts.rowid
               WHERE tweets_fts MATCH ?
                 AND tweets.account = ?
                 AND tweets.created_at >= ?
               ORDER BY tweets.created_at DESC""",
            (topic, account, since_date),
        ).fetchall()
    except Exception:
        # FTS not available — fallback to LIKE
        rows = conn.execute(
            "SELECT id, created_at, text FROM tweets "
            "WHERE account = ? AND text LIKE ? AND created_at >= ? "
            "ORDER BY created_at DESC",
            (account, f"%{topic}%", since_date),
        ).fetchall()
    return [(r[0], r[1], r[2]) for r in rows]
```

```python
def get_cache(topic, days=3, conn=None):
    close_after = conn is None
    if conn is None:
        conn = sqlite3.connect(DB_PATH)
    cutoff = (datetime.now() - timedelta(days=days)).isoformat()
    row = conn.execute(
        "SELECT result_json FROM query_cache WHERE topic = ? AND updated_at >= ?",
        (topic, cutoff),
    ).fetchone()
    if close_after:
        conn.close()
    return json.loads(row[0]) if row else None

def save_cache(topic, result_data, conn=None):
    close_after = conn is None
    if conn is None:
        conn = sqlite3.connect(DB_PATH)
    conn.execute(
        "INSERT OR REPLACE INTO query_cache (topic, result_json, updated_at) VALUES (?, ?, ?)",
        (topic, json.dumps(result_data), datetime.now().isoformat()),
    )
    conn.commit()
    if close_after:
        conn.close()
```

async def on_message(message):
    if message.author == bot.user:
        return
    await bot.process_commands(message)

    if message.content.startswith("$"):
        ticker = message.content[1:].upper().strip()
        if not (1 <= len(ticker) <= 10):
